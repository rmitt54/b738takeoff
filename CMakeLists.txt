cmake_minimum_required(VERSION 3.15)
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES "x86_64;arm64" CACHE STRING "")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
endif()
project(b738_takeoff_perf CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()

option(BUILD_PLUGIN "Build the X-Plane plugin (needs the X-Plane SDK)" ON)
option(BUILD_TESTS "Build the small test programs (no SDK needed)" OFF)
set(XPLANE_SDK "" CACHE PATH "Folder of the X-Plane SDK (contains CHeaders and Libraries)")

set(IMGUI_DIR ${CMAKE_SOURCE_DIR}/third_party/imgui)
set(CORE_SOURCES src/model.cpp src/airports.cpp src/airportdb_data.cpp)
set(IMGUI_SOURCES ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)

if(BUILD_PLUGIN)
  if(NOT XPLANE_SDK)
    message(FATAL_ERROR "Set -DXPLANE_SDK=<path to the unzipped X-Plane SDK>")
  endif()
  add_library(b738_takeoff_perf MODULE ${CORE_SOURCES} ${IMGUI_SOURCES} src/ui.cpp src/imgui_xp.cpp src/plugin.cpp)
  target_include_directories(b738_takeoff_perf PRIVATE src ${IMGUI_DIR} ${XPLANE_SDK}/CHeaders/XPLM)
  target_compile_definitions(b738_takeoff_perf PRIVATE XPLM200=1 XPLM210=1 XPLM300=1 XPLM301=1)
  if(WIN32)
    target_compile_definitions(b738_takeoff_perf PRIVATE IBM=1 NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(b738_takeoff_perf PRIVATE /utf-8)
    target_link_libraries(b738_takeoff_perf PRIVATE ${XPLANE_SDK}/Libraries/Win/XPLM_64.lib opengl32)
    set(OUT_NAME win)
  elseif(APPLE)
    target_compile_definitions(b738_takeoff_perf PRIVATE APL=1)
    target_link_libraries(b738_takeoff_perf PRIVATE "-F${XPLANE_SDK}/Libraries/Mac" "-framework XPLM" "-framework OpenGL")
    set(OUT_NAME mac)
  else()
    target_compile_definitions(b738_takeoff_perf PRIVATE LIN=1)
    target_link_options(b738_takeoff_perf PRIVATE -static-libgcc -static-libstdc++)
    set(OUT_NAME lin)
  endif()
  set(OUT_DIR ${CMAKE_BINARY_DIR}/b738_takeoff_perf/64)
  set_target_properties(b738_takeoff_perf PROPERTIES PREFIX "" SUFFIX ".xpl" OUTPUT_NAME ${OUT_NAME}
    LIBRARY_OUTPUT_DIRECTORY ${OUT_DIR} RUNTIME_OUTPUT_DIRECTORY ${OUT_DIR})
  foreach(cfg RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
    set_target_properties(b738_takeoff_perf PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${cfg} ${OUT_DIR}
      RUNTIME_OUTPUT_DIRECTORY_${cfg} ${OUT_DIR})
  endforeach()
endif()

if(BUILD_TESTS)
  add_executable(test_model test/test_model.cpp ${CORE_SOURCES})
  add_executable(test_ui test/test_ui.cpp src/ui.cpp ${CORE_SOURCES} ${IMGUI_SOURCES})
  target_include_directories(test_ui PRIVATE src ${IMGUI_DIR})
endif()
